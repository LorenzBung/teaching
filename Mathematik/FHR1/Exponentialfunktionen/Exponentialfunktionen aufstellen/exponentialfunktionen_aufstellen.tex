	\task{Funktionsgleichungen aus Graphen bestimmen}
	
	Bestimmen Sie die Funktionsgleichungen in der Form $a \cdot 3^{\pm x} + d$ der Graphen.
	
	\begin{plot}[xmin=-4, xmax=5, ymin=-3, ymax=4.5]
		\addplot[red, domain=-4:1]{4*3^x-3};
		\addplot[blue, domain=-4:1.5]{3^x+1};
		\addplot[Green, domain=-1:5]{2*3^(-x)};
		\addplot[cyan, domain=-4:1]{-3^x};
	\end{plot}
